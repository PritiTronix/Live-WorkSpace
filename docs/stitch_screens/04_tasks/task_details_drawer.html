export const validateTenantClaim = (payload: JwtPayload): boolean => {
  const { tenantId, tier, permissions } = payload;
  if (!tenantId || !isValidUUID(tenantId)) {
    throw new UnauthorizedException('Invalid tenant identifier');
  }
  return permissions.includes('tenant::read_write');
};
